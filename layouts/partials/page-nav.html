{{/* Prev/next navigation scoped to the current section. NOTE: Hugo's built-in .PrevInSection / .NextInSection follow Hugo's default DESCENDING sort order, which is inverted for sequential lessons (page 1's "next" would point backwards). So we compute position explicitly from the section's pages sorted ascending by weight. Set `weight` in each page's front matter to control order (10, 20, 30...). */}} {{ $pages := .CurrentSection.RegularPages.ByWeight }} {{ $currentURL := .RelPermalink }} {{ $prev := "" }} {{ $next := "" }} {{ range $i, $p := $pages }} {{ if eq $p.RelPermalink $currentURL }} {{ if gt $i 0 }} {{ $prev = index $pages (sub $i 1) }} {{ end }} {{ if lt (add $i 1) (len $pages) }} {{ $next = index $pages (add $i 1) }} {{ end }} {{ end }} {{ end }} {{ if or $prev $next }}
{{ with $prev }} ← Previous {{ .Title }} {{ end }}
{{ with $next }} Next → {{ .Title }} {{ end }}
{{ end }}
